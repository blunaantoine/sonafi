    95→// ---------------------------------------------------------------------------
    96→// localStorage helpers (sanitised — never trust what's in there)
    97→// ---------------------------------------------------------------------------
    98→
    99→function readDay(): DayNumber {
   100→  if (typeof window === "undefined") return 1;
   101→  try {
   102→    const raw = window.localStorage.getItem(LS_DAY);
   103→    const n = raw == null ? NaN : Number(JSON.parse(raw));
   104→    if (n === 1 || n === 2 || n === 3) return n;
   105→  } catch {
   106→    /* ignore */
   107→  }
   108→  return 1;
   109→}
   110→
   111→function readPause(): PauseType {
   112→  if (typeof window === "undefined") return "cafe";
   113→  try {
   114→    const raw = window.localStorage.getItem(LS_PAUSE);
   115→    if (raw == null) return "cafe";
   116→    const v = JSON.parse(raw) as string;
   117→    if (v === "cafe" || v === "dejeuner" || v === "jeux") return v;
   118→  } catch {
   119→    /* ignore */
   120→  }
   121→  return "cafe";
   122→}
   123→
   124→function writeLS(key: string, value: unknown): void {
   125→  try {
   126→    window.localStorage.setItem(key, JSON.stringify(value));
   127→  } catch {
   128→    /* ignore quota / privacy errors */
   129→  }
   130→}
   131→
   132→// ---------------------------------------------------------------------------
   133→// Friendly camera error translation
   134→// ---------------------------------------------------------------------------
   135→
   136→function friendlyCameraError(err: unknown): string {
   137→  const msg = err instanceof Error ? err.message : String(err);
   138→  if (/NotAllowedError|PermissionDenied/i.test(msg)) {
   139→    return "Accès à la caméra refusé. Autorisez la caméra dans les réglages du navigateur, puis réessayez.";
   140→  }
   141→  if (/NotFoundError|DevicesNotFoundError|CameraNotFound/i.test(msg)) {
   142→    return "Aucune caméra détectée sur cet appareil.";
   143→  }
   144→  if (/NotReadableError|TrackStartError|NotReadable/i.test(msg)) {
   145→    return "La caméra est déjà utilisée par une autre application. Fermez-la, puis réessayez.";
   146→  }
   147→  if (/InsecureContext|HTTPS|getUserMedia/i.test(msg)) {
   148→    return "La caméra nécessite une connexion HTTPS sécurisée.";
   149→  }
   150→  return "Impossible d'activer la caméra. Vérifiez votre navigateur et réessayez.";
   151→}

   153→// ---------------------------------------------------------------------------
   154→// Component
   155→// ---------------------------------------------------------------------------
   156→
   157→export function ScanTab() {
   158→  // ----- Persistent selectors ---------------------------------------------
   159→  const [day, setDay] = useState<DayNumber>(readDay);
   160→  const [pauseType, setPauseType] = useState<PauseType>(readPause);
   161→
   162→  useEffect(() => writeLS(LS_DAY, day), [day]);
   163→  useEffect(() => writeLS(LS_PAUSE, pauseType), [pauseType]);
   164→
   165→  // Refs mirroring day/pause so the (once-created) scan callback always sees
   166→  // the latest value even if the user switches mid-session.
   167→  const dayRef = useRef(day);
   168→  const pauseTypeRef = useRef(pauseType);
   169→  useEffect(() => {
   170→    dayRef.current = day;
   171→  }, [day]);
   172→  useEffect(() => {
   173→    pauseTypeRef.current = pauseType;
   174→  }, [pauseType]);
   175→
   176→  // ----- Camera state ------------------------------------------------------
   177→  const html5Ref = useRef<Html5Qrcode | null>(null);
   178→  const [cameraReady, setCameraReady] = useState(false);
   179→  const [cameraError, setCameraError] = useState<string | null>(null);
   180→  const [cameras, setCameras] = useState<{ id: string; label: string }[]>([]);
   181→  const [cameraIdx, setCameraIdx] = useState(0);
   182→
   183→  // ----- Scan flow state ---------------------------------------------------
   184→  const scanningRef = useRef(true);
   185→  const lastScanRef = useRef<{ id: string; ts: number }>({ id: "", ts: 0 });
   186→  const inFlightRef = useRef(false);
   187→  const resumeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   188→
   189→  const [overlay, setOverlay] = useState<OverlayState | null>(null);
   190→  const [recent, setRecent] = useState<ScanEntry[]>([]);
   191→  const [sessionCount, setSessionCount] = useState(0);
   192→
   193→  // ----- Pause / resume the camera whenever the overlay toggles -----------
   194→  useEffect(() => {
   195→    const inst = html5Ref.current;
   196→    if (!inst) return;
   197→    try {
   198→      if (overlay) inst.pause(true);
   199→      else inst.resume();
   200→    } catch {
   201→      /* ignore */
   202→    }
   203→  }, [overlay]);
   204→
   205→  // ----- recordPause: POST to the API, build overlay + recent entry -------
   206→  const recordPause = useCallback(async (participantId: string) => {
   207→    if (inFlightRef.current) return;
   208→    inFlightRef.current = true;
   209→    scanningRef.current = false;
   210→
   211→    const d = dayRef.current;
   212→    const p = pauseTypeRef.current;
   213→
   214→    // Show loading overlay immediately so the user sees feedback.
   215→    setOverlay({
   216→      state: "loading",
   217→      participantId,
   218→      day: d,
   219→      pauseType: p,
   220→    });
   221→
   222→    let result: OverlayState;
   223→    let status: ScanStatus;
   224→
   225→    try {
   226→      const res = await fetch("/api/sonafi/pause/record", {
   227→        method: "POST",
   228→        headers: { "Content-Type": "application/json" },
   229→        body: JSON.stringify({ participantId, day: d, pauseType: p }),
   230→      });
   231→      const data = (await res.json().catch(() => ({}))) as {
   232→        success?: boolean;
   233→        participant?: { id: string; name: string };
   234→        pause?: { timestamp: string };
   235→      };
   236→
   237→      if (res.ok && data?.success) {
   238→        result = {
   239→          state: "success",
   240→          participantId: data.participant?.id ?? participantId,
   241→          participantName: data.participant?.name,
   242→          day: d,
   243→          pauseType: p,
   244→          timestamp: data.pause?.timestamp,
   245→        };
   246→        status = "validated";
   247→        navigator.vibrate?.(120);
   248→      } else if (res.status === 409) {
   249→        result = {
   250→          state: "duplicate",
   251→          participantId: data.participant?.id ?? participantId,
   252→          participantName: data.participant?.name,
   253→          day: d,
   254→          pauseType: p,
   255→        };
   256→        status = "duplicate";
   257→        navigator.vibrate?.([60, 60, 60]);
   258→      } else if (res.status === 404) {
   259→        result = {
   260→          state: "unknown",
   261→          participantId,
   262→          day: d,
   263→          pauseType: p,
   264→        };
   265→        status = "unknown";
   266→        navigator.vibrate?.([60, 60, 60]);
   267→      } else {
   268→        result = { state: "error", participantId, day: d, pauseType: p };
   269→        status = "error";
   270→      }
   271→    } catch {
   272→      result = { state: "error", participantId, day: d, pauseType: p };
   273→      status = "error";
   274→    }
   275→
   276→    // Show the final result.
   277→    setOverlay(result);
   278→
   279→    // Push a recent-scans entry.
   280→    setRecent((prev) =>
   281→      [
   282→        {
   283→          id:
   284→            typeof crypto !== "undefined" && "randomUUID" in crypto
   285→              ? crypto.randomUUID()
   286→              : `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   287→          participantId: result.participantId,
   288→          participantName: result.participantName,
   289→          day: d,
   290→          pauseType: p,
   291→          timestamp: result.timestamp ?? new Date().toISOString(),
   292→          status,
   293→        } as ScanEntry,
   294→        ...prev,
   295→      ].slice(0, MAX_RECENT),
   296→    );
   297→    setSessionCount((c) => c + 1);
   298→
   299→    // Dismiss the overlay and resume scanning after OVERLAY_MS.
   300→    if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   301→    resumeTimerRef.current = setTimeout(() => {
   302→      setOverlay(null);
   303→      scanningRef.current = true;
   304→      inFlightRef.current = false;
   305→    }, OVERLAY_MS);
   306→  }, []);
   307→
   308→  // ----- handleScan: the callback passed to html5-qrcode ------------------
   309→  const handleScan = useCallback(
   310→    (decoded: string) => {
   311→      if (!scanningRef.current) return;
   312→      if (inFlightRef.current) return;
   313→      const now = Date.now();
   314→      const last = lastScanRef.current;
   315→      if (last.id === decoded && now - last.ts < DEBOUNCE_MS) return;
   316→      lastScanRef.current = { id: decoded, ts: now };
   317→      void recordPause(decoded);
   318→    },
   319→    [recordPause],
   320→  );
   321→
   322→  // ----- startCamera: dynamic-import html5-qrcode, prefer back camera -----
   323→  const startCamera = useCallback(
   324→    async (preferredIdx?: number) => {
   325→      setCameraError(null);
   326→      setCameraReady(false);
   327→
   328→      try {
   329→        const { Html5Qrcode } = await import("html5-qrcode");
   330→
   331→        // Tear down any previous instance before starting again.
   332→        if (html5Ref.current) {
   333→          try {
   334→            if (html5Ref.current.isScanning) await html5Ref.current.stop();
   335→          } catch {
   336→            /* ignore */
   337→          }
   338→          try {
   339→            html5Ref.current.clear();
   340→          } catch {
   341→            /* ignore */
   342→          }
   343→          html5Ref.current = null;
   344→        }
   345→
   346→        const instance = new Html5Qrcode(QR_READER_ID, {
   347→          verbose: false,
   348→        });
   349→        html5Ref.current = instance;
   350→
   351→        // Enumerate cameras (best-effort — falls back to facingMode).
   352→        let cams: { id: string; label: string }[] = [];
   353→        try {
   354→          cams = await Html5Qrcode.getCameras();
   355→          if (cams.length > 0) setCameras(cams);
   356→        } catch {
   357→          /* ignore — will use facingMode */
   358→        }
   359→
   360→        const idx =
   361→          preferredIdx != null ? preferredIdx : Math.min(cameraIdx, cams.length);
   362→        const chosen = cams[idx];
   363→
   364→        const config = {
   365→          fps: 10,
   366→          qrbox: (vw: number, vh: number) => {
   367→            const minEdge = Math.min(vw, vh);
   368→            const size = Math.max(160, Math.floor(minEdge * 0.7));
   369→            return { width: size, height: size };
   370→          },
   371→        };
   372→
   373→        const startConfig = chosen
   374→          ? { deviceId: { exact: chosen.id } }
   375→          : { facingMode: { ideal: "environment" as const } };
   376→
   377→        await instance.start(startConfig, config, handleScan, () => {
   378→          /* per-frame decode errors are noisy — ignore */
   379→        });
   380→
   381→        setCameraReady(true);
   382→      } catch (err) {
   383→        setCameraError(friendlyCameraError(err));
   384→        setCameraReady(false);
   385→      }
   386→    },
   387→    [cameraIdx, handleScan],
   388→  );
   389→
   390→  // ----- mount: auto-start camera; unmount: stop + clear -------------------
   391→  useEffect(() => {
   392→    void startCamera();
   393→    return () => {
   394→      if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   395→      const inst = html5Ref.current;
   396→      html5Ref.current = null;
   397→      scanningRef.current = false;
   398→      inFlightRef.current = false;
   399→      void (async () => {
   400→        if (!inst) return;
   401→        try {
   402→          if (inst.isScanning) await inst.stop();
   403→        } catch {
   404→          /* ignore */
   405→        }
   406→        try {
   407→          inst.clear();
   408→        } catch {
   409→          /* ignore */
   410→        }
   411→      })();
   412→    };
   413→    // We deliberately start only once on mount — startCamera is captured at
   414→    // first render, which is exactly what we want (initial camera).
   415→  }, []);
   416→
   417→  // ----- cycleCamera: switch to next available camera ---------------------
   418→  const cycleCamera = useCallback(async () => {
   419→    if (cameras.length < 2) return;
   420→    const next = (cameraIdx + 1) % cameras.length;
   421→    setCameraIdx(next);
   422→    await startCamera(next);
   423→  }, [cameras.length, cameraIdx, startCamera]);
   424→
   425→  // -----------------------------------------------------------------------
   426→  // Render
   427→  // -----------------------------------------------------------------------
   428→  return (
   429→    <div className="flex flex-col gap-5">
   430→      {/* Day selector */}
   431→      <section aria-label="Choix du jour">
   432→        <h2 className="mb-2 text-sm font-semibold text-muted-foreground">Jour</h2>
   433→        <div className="grid grid-cols-3 gap-2">
   434→          {DAYS.map((d) => {
   435→            const active = d === day;
   436→            return (
   437→              <button
   438→                key={d}
   439→                type="button"
   440→                onClick={() => setDay(d)}
   441→                aria-pressed={active}
   442→                className={cn(
   443→                  "flex h-12 items-center justify-center rounded-xl border text-sm font-semibold transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400 focus-visible:ring-offset-2",
   444→                  active
   445→                    ? "border-emerald-500 bg-emerald-500 text-white shadow-sm"
   446→                    : "border-border bg-background text-foreground hover:bg-emerald-50",
   447→                )}
   448→              >
   449→                {DAY_LABELS[d]}
   450→              </button>
   451→            );
   452→          })}
   453→        </div>
   454→      </section>
   455→
   456→      {/* Pause selector */}
   457→      <section aria-label="Choix de la pause">
   458→        <h2 className="mb-2 text-sm font-semibold text-muted-foreground">Pause</h2>
   459→        <div className="grid grid-cols-3 gap-2">
   460→          {PAUSE_TYPES.map((p) => {
   461→            const meta = PAUSE_META[p];
   462→            const active = p === pauseType;
   463→            return (
   464→              <button
   465→                key={p}
   466→                type="button"
   467→                onClick={() => setPauseType(p)}
   468→                aria-pressed={active}
   469→                className={cn(
   470→                  "flex h-16 flex-col items-center justify-center gap-1 rounded-xl border text-xs font-semibold transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-2",
   471→                  active
   472→                    ? cn(meta.soft, "ring-2", meta.ring, "focus-visible:ring-emerald-400")
   473→                    : "border-border bg-background text-foreground hover:bg-accent focus-visible:ring-emerald-400",
   474→                )}
   475→              >
   476→                <span className="text-2xl leading-none" aria-hidden>
   477→                  {meta.emoji}
   478→                </span>
   479→                <span className="leading-tight">
   480→                  {meta.label.replace(/^Pause\s+/, "")}
   481→                </span>
   482→              </button>
   483→            );
   484→          })}
   485→        </div>
   486→      </section>
   487→
   488→      {/* Camera */}
   489→      <section aria-label="Caméra de scan">
   490→        <div className="mb-2 flex items-center justify-between gap-3">
   491→          <h2 className="text-sm font-semibold text-muted-foreground">Caméra</h2>
   492→          <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground">
   493→            <ScanLine className="h-3.5 w-3.5" />
   494→            {sessionCount} scan{sessionCount === 1 ? "" : "s"} cette session
   495→          </span>
   496→        </div>
   497→
   498→        <Card className="overflow-hidden border-emerald-100 py-0">
   499→          <CardContent className="p-3 sm:p-4">
   500→            <div className="relative mx-auto aspect-square w-full max-w-md overflow-hidden rounded-2xl border border-border bg-slate-900">
   501→              {/* html5-qrcode injects the <video> here */}
   502→              <div
   503→                id={QR_READER_ID}
   504→                className="absolute inset-0 [&>video]:h-full [&>video]:w-full [&>video]:object-cover [&>video]:bg-slate-900"
   505→              />
   506→
   507→              {/* Scan reticle overlay */}
   508→              {cameraReady && !cameraError && (
   509→                <div className="pointer-events-none absolute inset-0 flex items-center justify-center">
   510→                  <div className="relative h-[62%] w-[62%]">
   511→                    <span className="absolute left-0 top-0 h-8 w-8 rounded-tl-lg border-l-4 border-t-4 border-white/90" />
   512→                    <span className="absolute right-0 top-0 h-8 w-8 rounded-tr-lg border-r-4 border-t-4 border-white/90" />
   513→                    <span className="absolute bottom-0 left-0 h-8 w-8 rounded-bl-lg border-b-4 border-l-4 border-white/90" />
   514→                    <span className="absolute bottom-0 right-0 h-8 w-8 rounded-br-lg border-b-4 border-r-4 border-white/90" />
   515→                    <motion.span
   516→                      className="absolute inset-x-4 top-1/2 h-0.5 rounded-full bg-emerald-400 shadow-[0_0_8px_2px_rgba(16,185,129,0.65)]"
   517→                      animate={{ y: [-90, 90, -90] }}
   518→                      transition={{
   519→                        duration: 2.4,
   520→                        repeat: Infinity,
   521→                        ease: "easeInOut",
   522→                      }}
   523→                    />
   524→                  </div>
   525→                </div>
   526→              )}
   527→
   528→              {/* Initialising spinner */}
   529→              {!cameraReady && !cameraError && (
   530→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 text-white/90">
   531→                  <Loader2 className="h-8 w-8 animate-spin" />
   532→                  <p className="text-sm">Activation de la caméra…</p>
   533→                </div>
   534→              )}
   535→
   536→              {/* Camera error card */}
   537→              {cameraError && (
   538→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 bg-slate-900/95 p-6 text-center text-white">
   539→                  <AlertTriangle className="h-10 w-10 text-rose-300" />
   540→                  <p className="max-w-xs text-sm leading-relaxed text-white/90">
   541→                    {cameraError}
   542→                  </p>
   543→                  <Button
   544→                    type="button"
   545→                    size="sm"
   546→                    onClick={() => void startCamera()}
   547→                    className="bg-emerald-500 text-white hover:bg-emerald-600"
   548→                  >
   549→                    <RefreshCw className="h-4 w-4" />
   550→                    Réessayer
   551→                  </Button>
   552→                </div>
   553→              )}
   554→
   555→              {/* Camera switch */}
   556→              {cameraReady && cameras.length > 1 && (
   557→                <Button
   558→                  type="button"
   559→                  size="sm"
   560→                  variant="secondary"
   561→                  onClick={() => void cycleCamera()}
   562→                  className="absolute right-2 top-2 z-10 h-8 gap-1 bg-black/55 text-white backdrop-blur hover:bg-black/70"
   563→                >
   564→                  <RotateCcw className="h-3.5 w-3.5" />
   565→                  Caméra
   566→                </Button>
   567→              )}
   568→            </div>
   569→
   570→            {/* Status row */}
   571→            <div className="mt-3 flex items-center justify-between text-xs text-muted-foreground">
   572→              <span className="inline-flex items-center gap-1.5">
   573→                <span
   574→                  className={cn(
   575→                    "h-2 w-2 rounded-full",
   576→                    cameraReady
   577→                      ? "animate-pulse bg-emerald-500"
   578→                      : "bg-muted-foreground/40",
   579→                  )}
   580→                />
   581→                {cameraReady
   582→                  ? "Caméra active"
   583→                  : cameraError
   584→                    ? "Caméra indisponible"
   585→                    : "Démarrage…"}
   586→              </span>
   587→              <span className="inline-flex items-center gap-1.5">
   588→                {PAUSE_META[pauseType].emoji} {PAUSE_META[pauseType].label} ·{" "}
   589→                {DAY_LABELS[day]}
   590→              </span>
   591→            </div>
   592→          </CardContent>
   593→        </Card>
   594→      </section>
   595→
   596→      {/* Recent scans */}
   597→      <section aria-label="Scans récents">
   598→        <div className="mb-2 flex items-center justify-between">
   599→          <h2 className="text-sm font-semibold text-muted-foreground">
   600→            Scans récents
   601→          </h2>
   602→          <span className="text-xs text-muted-foreground">
   603→            {recent.length}/{MAX_RECENT}
   604→          </span>
   605→        </div>
   606→
   607→        {recent.length === 0 ? (
   608→          <Card className="border-dashed bg-muted/30 py-0">
   609→            <CardContent className="flex items-center justify-center py-8 text-sm text-muted-foreground">
   610→              Aucun scan pour le moment
   611→            </CardContent>
   612→          </Card>
   613→        ) : (
   614→          <div
   615→            className="max-h-72 overflow-y-auto pr-1
   616→              [&::-webkit-scrollbar]:w-2
   617→              [&::-webkit-scrollbar-thumb]:rounded-full
   618→              [&::-webkit-scrollbar-thumb]:bg-muted-foreground/30
   619→              [&::-webkit-scrollbar-track]:bg-transparent"
   620→          >
   621→            <ul className="flex flex-col gap-2">
   622→              {recent.map((e) => (
   623→                <li key={e.id}>
   624→                  <Card className="border-border/70 py-0">
   625→                    <CardContent className="flex items-center gap-3 px-3 py-2.5">
   626→                      <span
   627→                        className={cn(
   628→                          "h-2.5 w-2.5 shrink-0 rounded-full",
   629→                          PAUSE_DOT[e.pauseType],
   630→                        )}
   631→                        aria-hidden
   632→                      />
   633→                      <div className="flex min-w-0 flex-1 flex-col">
   634→                        <div className="flex items-baseline gap-2">
   635→                          <span className="truncate text-sm font-semibold">
   636→                            {e.participantName ?? "Badge non reconnu"}
   637→                          </span>
   638→                          <span className="shrink-0 font-mono text-xs text-muted-foreground">
   639→                            {e.participantId}
   640→                          </span>
   641→                        </div>
   642→                        <div className="truncate text-xs text-muted-foreground">
   643→                          {DAY_LABELS[e.day]} · {PAUSE_META[e.pauseType].label} ·{" "}
   644→                          {formatTimestamp(e.timestamp)}
   645→                        </div>
   646→                      </div>
   647→                      <StatusPill status={e.status} />
   648→                    </CardContent>
   649→                  </Card>
   650→                </li>
   651→              ))}
   652→            </ul>
   653→          </div>
   654→        )}
   655→      </section>
   656→
   657→      {/* Full-screen result overlay */}
   658→      <ResultOverlay overlay={overlay} />
   659→    </div>
   660→  );
   661→}

   669→function StatusPill({ status }: { status: ScanStatus }) {
   670→  const map: Record<ScanStatus, { label: string; className: string }> = {

   672→      label: "Validé",
   673→      className: "border-emerald-200 bg-emerald-100 text-emerald-800",

   675→    duplicate: {
   676→      label: "Doublon",
   677→      className: "border-rose-200 bg-rose-100 text-rose-800",
   678→    },
   679→    unknown: {
   680→      label: "Inconnu",
   681→      className: "border-amber-200 bg-amber-100 text-amber-800",
   682→    },
   683→    error: {
   684→      label: "Erreur",
   685→      className: "border-slate-200 bg-slate-100 text-slate-700",
   686→    },
   687→  };

   701→// ---------------------------------------------------------------------------
   702→// ResultOverlay — full-screen verdict shown after each scan
   703→// ---------------------------------------------------------------------------
   704→
   705→function ResultOverlay({ overlay }: { overlay: OverlayState | null }) {
   706→  const theme: OverlayTheme | null = (() => {
   707→    if (!overlay) return null;
   708→    const meta = `${PAUSE_META[overlay.pauseType].emoji} ${PAUSE_META[overlay.pauseType].label} · ${DAY_LABELS[overlay.day]}`;
   709→    switch (overlay.state) {
   710→      case "loading":
   711→        return {
   712→          bg: "bg-slate-900/95",
   713→          icon: <Loader2 className="h-16 w-16 animate-spin text-white" />,
   714→          headline: "Vérification…",
   715→          sub: overlay.participantId,
   716→        };
   717→      case "success":
   718→        return {
   719→          bg: "bg-emerald-600",
   720→          icon: <Check className="h-20 w-20 text-white" strokeWidth={3} />,
   721→          headline: "Pause validée",
   722→          sub: overlay.participantName ?? "Participant",
   723→          meta,
   724→        };
   725→      case "duplicate":
   726→        return {
   727→          bg: "bg-rose-600",
   728→          icon: <X className="h-20 w-20 text-white" strokeWidth={3} />,
   729→          headline: "Déjà servi(e)",
   730→          sub: overlay.participantName ?? overlay.participantId,
   731→          meta,
   732→        };
   733→      case "unknown":
   734→        return {
   735→          bg: "bg-amber-500",
   736→          icon: <AlertTriangle className="h-20 w-20 text-white" strokeWidth={2.5} />,
   737→          headline: "Badge non reconnu",
   738→          sub: overlay.participantId,
   739→        };
   740→      case "error":
   741→        return {
   742→          bg: "bg-slate-700",
   743→          icon: <WifiOff className="h-20 w-20 text-white" strokeWidth={2.5} />,
   744→          headline: "Erreur, réessayez",
   745→          sub: "Vérifiez votre connexion",
   746→        };
   747→    }
   748→  })();
   749→
   750→  return (
   751→    <AnimatePresence>
   752→      {overlay && theme && (
   753→        <motion.div
   754→          initial={{ opacity: 0 }}
   755→          animate={{ opacity: 1 }}
   756→          exit={{ opacity: 0 }}
   757→          transition={{ duration: 0.18 }}
   758→          className={cn(
   759→            "fixed inset-0 z-50 flex flex-col items-center justify-center p-8 text-center text-white",
   760→            theme.bg,
   761→          )}
   762→          aria-live="assertive"
   763→          role="status"
   764→        >
   765→          <motion.div
   766→            key={overlay.state}
   767→            initial={{ scale: 0.5, opacity: 0 }}
   768→            animate={{ scale: 1, opacity: 1 }}
   769→            transition={{ type: "spring", stiffness: 320, damping: 22 }}
   770→            className="flex flex-col items-center gap-5"
   771→          >
   772→            {theme.icon}
   773→            <h2 className="text-3xl font-bold tracking-tight sm:text-4xl">
   774→              {theme.headline}
   775→            </h2>
   776→            {theme.sub && (
   777→              <p className="max-w-md break-words text-xl font-medium opacity-95">
   778→                {theme.sub}
   779→              </p>
   780→            )}
   781→            {theme.meta && <p className="text-lg opacity-90">{theme.meta}</p>}
   782→          </motion.div>
   783→        </motion.div>
   784→      )}
   785→    </AnimatePresence>
   786→  );
   787→}
