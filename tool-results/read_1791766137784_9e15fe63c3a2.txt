    98→    98→function readDay(): DayNumber {
    99→    99→  if (typeof window === "undefined") return 1;
   100→   100→  try {
   101→   101→    const raw = window.localStorage.getItem(LS_DAY);
   102→   102→    const n = raw == null ? NaN : Number(JSON.parse(raw));
   103→   103→    if (n === 1 || n === 2 || n === 3) return n;
   104→   104→  } catch {
   105→   105→    /* ignore */
   106→   106→  }
   107→   107→  return 1;
   108→   108→}
   109→   109→
   110→   110→function readPause(): PauseType {
   111→   111→  if (typeof window === "undefined") return "cafe";
   112→   112→  try {
   113→   113→    const raw = window.localStorage.getItem(LS_PAUSE);
   114→   114→    if (raw == null) return "cafe";
   115→   115→    const v = JSON.parse(raw) as string;
   116→   116→    if (v === "cafe" || v === "dejeuner") return v;
   117→   117→  } catch {
   118→   118→    /* ignore */
   119→   119→  }
   120→   120→  return "cafe";
   121→   121→}
   122→   122→
   123→   123→function writeLS(key: string, value: unknown): void {
   124→   124→  try {
   125→   125→    window.localStorage.setItem(key, JSON.stringify(value));
   126→   126→  } catch {
   127→   127→    /* ignore quota / privacy errors */
   128→   128→  }
   129→   129→}

   131→   131→// ---------------------------------------------------------------------------
   132→   132→// Friendly camera error translation
   133→   133→// ---------------------------------------------------------------------------
   134→   134→
   135→   135→function friendlyCameraError(err: unknown): string {
   136→   136→  const msg = err instanceof Error ? err.message : String(err);
   137→   137→  if (/NotAllowedError|PermissionDenied/i.test(msg)) {
   138→   138→    return "Accès à la caméra refusé. Autorisez la caméra dans les réglages du navigateur, puis réessayez.";
   139→   139→  }
   140→   140→  if (/NotFoundError|DevicesNotFoundError|CameraNotFound/i.test(msg)) {
   141→   141→    return "Aucune caméra détectée sur cet appareil.";
   142→   142→  }
   143→   143→  if (/NotReadableError|TrackStartError|NotReadable/i.test(msg)) {
   144→   144→    return "La caméra est déjà utilisée par une autre application. Fermez-la, puis réessayez.";
   145→   145→  }
   146→   146→  if (/InsecureContext|HTTPS|getUserMedia/i.test(msg)) {
   147→   147→    return "La caméra nécessite une connexion HTTPS sécurisée.";
   148→   148→  }
   149→   149→  return "Impossible d'activer la caméra. Vérifiez votre navigateur et réessayez.";
   150→   150→}

   152→   152→// ---------------------------------------------------------------------------
   153→   153→// Component
   154→   154→// ---------------------------------------------------------------------------
   155→   155→
   156→   156→export function ScanTab() {
   157→   157→  // ----- Persistent selectors ---------------------------------------------
   158→   158→  const [day, setDay] = useState<DayNumber>(readDay);
   159→   159→  const [pauseType, setPauseType] = useState<PauseType>(readPause);
   160→   160→
   161→   161→  useEffect(() => writeLS(LS_DAY, day), [day]);
   162→   162→  useEffect(() => writeLS(LS_PAUSE, pauseType), [pauseType]);
   163→   163→
   164→   164→  // Refs mirroring day/pause so the (once-created) scan callback always sees
   165→   165→  // the latest value even if the user switches mid-session.
   166→   166→  const dayRef = useRef(day);
   167→   167→  const pauseTypeRef = useRef(pauseType);
   168→   168→  useEffect(() => {
   169→   169→    dayRef.current = day;
   170→   170→  }, [day]);
   171→   171→  useEffect(() => {
   172→   172→    pauseTypeRef.current = pauseType;
   173→   173→  }, [pauseType]);
   174→   174→
   175→   175→  // ----- Camera state ------------------------------------------------------
   176→   176→  const html5Ref = useRef<Html5Qrcode | null>(null);
   177→   177→  const [cameraReady, setCameraReady] = useState(false);
   178→   178→  const [cameraError, setCameraError] = useState<string | null>(null);
   179→   179→  const [cameras, setCameras] = useState<{ id: string; label: string }[]>([]);
   180→   180→  const [cameraIdx, setCameraIdx] = useState(0);
   181→   181→
   182→   182→  // ----- Scan flow state ---------------------------------------------------
   183→   183→  const scanningRef = useRef(true);
   184→   184→  const lastScanRef = useRef<{ id: string; ts: number }>({ id: "", ts: 0 });
   185→   185→  const inFlightRef = useRef(false);
   186→   186→  const resumeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   187→   187→
   188→   188→  const [overlay, setOverlay] = useState<OverlayState | null>(null);
   189→   189→  const [recent, setRecent] = useState<ScanEntry[]>([]);
   190→   190→  const [sessionCount, setSessionCount] = useState(0);
   191→   191→
   192→   192→  // ----- Pause / resume the camera whenever the overlay toggles -----------
   193→   193→  useEffect(() => {
   194→   194→    const inst = html5Ref.current;
   195→   195→    if (!inst) return;
   196→   196→    try {
   197→   197→      if (overlay) inst.pause(true);
   198→   198→      else inst.resume();
   199→   199→    } catch {
   200→   200→      /* ignore */
   201→   201→    }
   202→   202→  }, [overlay]);
   203→   203→
   204→   204→  // ----- recordPause: POST to the API, build overlay + recent entry -------
   205→   205→  const recordPause = useCallback(async (participantId: string) => {
   206→   206→    if (inFlightRef.current) return;
   207→   207→    inFlightRef.current = true;
   208→   208→    scanningRef.current = false;
   209→   209→
   210→   210→    const d = dayRef.current;
   211→   211→    const p = pauseTypeRef.current;
   212→   212→
   213→   213→    // Show loading overlay immediately so the user sees feedback.
   214→   214→    setOverlay({
   215→   215→      state: "loading",
   216→   216→      participantId,
   217→   217→      day: d,
   218→   218→      pauseType: p,
   219→   219→    });
   220→   220→
   221→   221→    let result: OverlayState;
   222→   222→    let status: ScanStatus;
   223→   223→
   224→   224→    try {
   225→   225→      const res = await fetch("/api/sonafi/pause/record", {
   226→   226→        method: "POST",
   227→   227→        headers: { "Content-Type": "application/json" },
   228→   228→        body: JSON.stringify({ participantId, day: d, pauseType: p }),
   229→   229→      });
   230→   230→      const data = (await res.json().catch(() => ({}))) as {
   231→   231→        success?: boolean;
   232→   232→        participant?: { id: string; name: string };
   233→   233→        pause?: { timestamp: string };
   234→   234→      };
   235→   235→
   236→   236→      if (res.ok && data?.success) {
   237→   237→        result = {
   238→   238→          state: "success",
   239→   239→          participantId: data.participant?.id ?? participantId,
   240→   240→          participantName: data.participant?.name,
   241→   241→          day: d,
   242→   242→          pauseType: p,
   243→   243→          timestamp: data.pause?.timestamp,
   244→   244→        };
   245→   245→        status = "validated";
   246→   246→        navigator.vibrate?.(120);
   247→   247→      } else if (res.status === 409) {
   248→   248→        result = {
   249→   249→          state: "duplicate",
   250→   250→          participantId: data.participant?.id ?? participantId,
   251→   251→          participantName: data.participant?.name,
   252→   252→          day: d,
   253→   253→          pauseType: p,
   254→   254→        };
   255→   255→        status = "duplicate";
   256→   256→        navigator.vibrate?.([60, 60, 60]);
   257→   257→      } else if (res.status === 404) {
   258→   258→        result = {
   259→   259→          state: "unknown",
   260→   260→          participantId,
   261→   261→          day: d,
   262→   262→          pauseType: p,
   263→   263→        };
   264→   264→        status = "unknown";
   265→   265→        navigator.vibrate?.([60, 60, 60]);
   266→   266→      } else {
   267→   267→        result = { state: "error", participantId, day: d, pauseType: p };
   268→   268→        status = "error";
   269→   269→      }
   270→   270→    } catch {
   271→   271→      result = { state: "error", participantId, day: d, pauseType: p };
   272→   272→      status = "error";
   273→   273→    }
   274→   274→
   275→   275→    // Show the final result.
   276→   276→    setOverlay(result);
   277→   277→
   278→   278→    // Push a recent-scans entry.
   279→   279→    setRecent((prev) =>
   280→   280→      [
   281→   281→        {
   282→   282→          id:
   283→   283→            typeof crypto !== "undefined" && "randomUUID" in crypto
   284→   284→              ? crypto.randomUUID()
   285→   285→              : `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   286→   286→          participantId: result.participantId,
   287→   287→          participantName: result.participantName,
   288→   288→          day: d,
   289→   289→          pauseType: p,
   290→   290→          timestamp: result.timestamp ?? new Date().toISOString(),
   291→   291→          status,
   292→   292→        } as ScanEntry,
   293→   293→        ...prev,
   294→   294→      ].slice(0, MAX_RECENT),
   295→   295→    );
   296→   296→    setSessionCount((c) => c + 1);
   297→   297→
   298→   298→    // Dismiss the overlay and resume scanning after OVERLAY_MS.
   299→   299→    if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   300→   300→    resumeTimerRef.current = setTimeout(() => {
   301→   301→      setOverlay(null);
   302→   302→      scanningRef.current = true;
   303→   303→      inFlightRef.current = false;
   304→   304→    }, OVERLAY_MS);
   305→   305→  }, []);
   306→   306→
   307→   307→  // ----- handleScan: the callback passed to html5-qrcode ------------------
   308→   308→  const handleScan = useCallback(
   309→   309→    (decoded: string) => {
   310→   310→      if (!scanningRef.current) return;
   311→   311→      if (inFlightRef.current) return;
   312→   312→      const now = Date.now();
   313→   313→      const last = lastScanRef.current;
   314→   314→      if (last.id === decoded && now - last.ts < DEBOUNCE_MS) return;
   315→   315→      lastScanRef.current = { id: decoded, ts: now };
   316→   316→      void recordPause(decoded);
   317→   317→    },
   318→   318→    [recordPause],
   319→   319→  );
   320→   320→
   321→   321→  // ----- startCamera: dynamic-import html5-qrcode, prefer back camera -----
   322→   322→  const startCamera = useCallback(
   323→   323→    async (preferredIdx?: number) => {
   324→   324→      setCameraError(null);
   325→   325→      setCameraReady(false);
   326→   326→
   327→   327→      try {
   328→   328→        const { Html5Qrcode } = await import("html5-qrcode");
   329→   329→
   330→   330→        // Tear down any previous instance before starting again.
   331→   331→        if (html5Ref.current) {
   332→   332→          try {
   333→   333→            if (html5Ref.current.isScanning) await html5Ref.current.stop();
   334→   334→          } catch {
   335→   335→            /* ignore */
   336→   336→          }
   337→   337→          try {
   338→   338→            html5Ref.current.clear();
   339→   339→          } catch {
   340→   340→            /* ignore */
   341→   341→          }
   342→   342→          html5Ref.current = null;
   343→   343→        }
   344→   344→
   345→   345→        const instance = new Html5Qrcode(QR_READER_ID, {
   346→   346→          verbose: false,
   347→   347→        });
   348→   348→        html5Ref.current = instance;
   349→   349→
   350→   350→        // Enumerate cameras (best-effort — falls back to facingMode).
   351→   351→        let cams: { id: string; label: string }[] = [];
   352→   352→        try {
   353→   353→          cams = await Html5Qrcode.getCameras();
   354→   354→          if (cams.length > 0) setCameras(cams);
   355→   355→        } catch {
   356→   356→          /* ignore — will use facingMode */
   357→   357→        }
   358→   358→
   359→   359→        const idx =
   360→   360→          preferredIdx != null ? preferredIdx : Math.min(cameraIdx, cams.length);
   361→   361→        const chosen = cams[idx];
   362→   362→
   363→   363→        const config = {
   364→   364→          fps: 10,
   365→   365→          qrbox: (vw: number, vh: number) => {
   366→   366→            const minEdge = Math.min(vw, vh);
   367→   367→            const size = Math.max(160, Math.floor(minEdge * 0.7));
   368→   368→            return { width: size, height: size };
   369→   369→          },
   370→   370→        };
   371→   371→
   372→   372→        const startConfig = chosen
   373→   373→          ? { deviceId: { exact: chosen.id } }
   374→   374→          : { facingMode: { ideal: "environment" as const } };
   375→   375→
   376→   376→        await instance.start(startConfig, config, handleScan, () => {
   377→   377→          /* per-frame decode errors are noisy — ignore */
   378→   378→        });
   379→   379→
   380→   380→        setCameraReady(true);
   381→   381→      } catch (err) {
   382→   382→        setCameraError(friendlyCameraError(err));
   383→   383→        setCameraReady(false);
   384→   384→      }
   385→   385→    },
   386→   386→    [cameraIdx, handleScan],
   387→   387→  );
   388→   388→
   389→   389→  // ----- mount: auto-start camera; unmount: stop + clear -------------------
   390→   390→  useEffect(() => {
   391→   391→    void startCamera();
   392→   392→    return () => {
   393→   393→      if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   394→   394→      const inst = html5Ref.current;
   395→   395→      html5Ref.current = null;
   396→   396→      scanningRef.current = false;
   397→   397→      inFlightRef.current = false;
   398→   398→      void (async () => {
   399→   399→        if (!inst) return;
   400→   400→        try {
   401→   401→          if (inst.isScanning) await inst.stop();
   402→   402→        } catch {
   403→   403→          /* ignore */
   404→   404→        }
   405→   405→        try {
   406→   406→          inst.clear();
   407→   407→        } catch {
   408→   408→          /* ignore */
   409→   409→        }
   410→   410→      })();
   411→   411→    };
   412→   412→    // We deliberately start only once on mount — startCamera is captured at
   413→   413→    // first render, which is exactly what we want (initial camera).
   414→   414→  }, []);
   415→   415→
   416→   416→  // ----- cycleCamera: switch to next available camera ---------------------
   417→   417→  const cycleCamera = useCallback(async () => {
   418→   418→    if (cameras.length < 2) return;
   419→   419→    const next = (cameraIdx + 1) % cameras.length;
   420→   420→    setCameraIdx(next);
   421→   421→    await startCamera(next);
   422→   422→  }, [cameras.length, cameraIdx, startCamera]);
   423→   423→
   424→   424→  // -----------------------------------------------------------------------
   425→   425→  // Render
   426→   426→  // -----------------------------------------------------------------------
   427→   427→  return (
   428→   428→    <div className="flex flex-col gap-5">
   429→   429→      {/* Day selector */}
   430→   430→      <section aria-label="Choix du jour">
   431→   431→        <h2 className="mb-2 text-sm font-semibold text-muted-foreground">Jour</h2>
   432→   432→        <div className="grid grid-cols-3 gap-2">
   433→   433→          {DAYS.map((d) => {
   434→   434→            const active = d === day;
   435→   435→            return (
   436→   436→              <button
   437→   437→                key={d}
   438→   438→                type="button"
   439→   439→                onClick={() => setDay(d)}
   440→   440→                aria-pressed={active}
   441→   441→                className={cn(
   442→   442→                  "flex h-12 items-center justify-center rounded-xl border text-sm font-semibold transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400 focus-visible:ring-offset-2",
   443→   443→                  active
   444→   444→                    ? "border-emerald-500 bg-emerald-500 text-white shadow-sm"
   445→   445→                    : "border-border bg-background text-foreground hover:bg-emerald-50",
   446→   446→                )}
   447→   447→              >
   448→   448→                {DAY_LABELS[d]}
   449→   449→              </button>
   450→   450→            );
   451→   451→          })}
   452→   452→        </div>
   453→   453→      </section>
   454→   454→
   455→   455→      {/* Pause selector */}
   456→   456→      <section aria-label="Choix de la pause">
   457→   457→        <h2 className="mb-2 text-sm font-semibold text-muted-foreground">Pause</h2>
   458→   458→        <div className="grid grid-cols-3 gap-2">
   459→   459→          {PAUSE_TYPES.map((p) => {
   460→   460→            const meta = PAUSE_META[p];
   461→   461→            const active = p === pauseType;
   462→   462→            return (
   463→   463→              <button
   464→   464→                key={p}
   465→   465→                type="button"
   466→   466→                onClick={() => setPauseType(p)}
   467→   467→                aria-pressed={active}
   468→   468→                className={cn(
   469→   469→                  "flex h-16 flex-col items-center justify-center gap-1 rounded-xl border text-xs font-semibold transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-2",
   470→   470→                  active
   471→   471→                    ? cn(meta.soft, "ring-2", meta.ring, "focus-visible:ring-emerald-400")
   472→   472→                    : "border-border bg-background text-foreground hover:bg-accent focus-visible:ring-emerald-400",
   473→   473→                )}
   474→   474→              >
   475→   475→                <span className="text-2xl leading-none" aria-hidden>
   476→   476→                  {meta.emoji}
   477→   477→                </span>
   478→   478→                <span className="leading-tight">
   479→   479→                  {meta.label.replace(/^Pause\s+/, "")}
   480→   480→                </span>
   481→   481→              </button>
   482→   482→            );
   483→   483→          })}
   484→   484→        </div>
   485→   485→      </section>
   486→   486→
   487→   487→      {/* Camera */}
   488→   488→      <section aria-label="Caméra de scan">
   489→   489→        <div className="mb-2 flex items-center justify-between gap-3">
   490→   490→          <h2 className="text-sm font-semibold text-muted-foreground">Caméra</h2>
   491→   491→          <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground">
   492→   492→            <ScanLine className="h-3.5 w-3.5" />
   493→   493→            {sessionCount} scan{sessionCount === 1 ? "" : "s"} cette session
   494→   494→          </span>
   495→   495→        </div>
   496→   496→
