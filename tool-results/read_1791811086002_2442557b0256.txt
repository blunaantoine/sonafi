   100→   100→   100→
   101→   101→   101→function readDay(): DayNumber {
   102→   102→   102→  if (typeof window === "undefined") return 1;
   103→   103→   103→  try {
   104→   104→   104→    const raw = window.localStorage.getItem(LS_DAY);
   105→   105→   105→    const n = raw == null ? NaN : Number(JSON.parse(raw));
   106→   106→   106→    if (n === 1 || n === 2 || n === 3) return n;
   107→   107→   107→  } catch {
   108→   108→   108→    /* ignore */
   109→   109→   109→  }
   110→   110→   110→  return 1;
   111→   111→   111→}
   112→   112→   112→
   113→   113→   113→function writeLS(key: string, value: unknown): void {
   114→   114→   114→  try {
   115→   115→   115→    window.localStorage.setItem(key, JSON.stringify(value));
   116→   116→   116→  } catch {
   117→   117→   117→    /* ignore quota / privacy errors */
   118→   118→   118→  }
   119→   119→   119→}

   124→   124→   124→
   125→   125→   125→function friendlyCameraError(err: unknown): string {
   126→   126→   126→  const msg = err instanceof Error ? err.message : String(err);
   127→   127→   127→  if (/NotAllowedError|PermissionDenied/i.test(msg)) {
   128→   128→   128→    return "Accès à la caméra refusé. Autorisez la caméra dans les réglages du navigateur, puis réessayez.";
   129→   129→   129→  }
   130→   130→   130→  if (/NotFoundError|DevicesNotFoundError|CameraNotFound/i.test(msg)) {
   131→   131→   131→    return "Aucune caméra détectée sur cet appareil.";
   132→   132→   132→  }
   133→   133→   133→  if (/NotReadableError|TrackStartError|NotReadable/i.test(msg)) {
   134→   134→   134→    return "La caméra est déjà utilisée par une autre application. Fermez-la, puis réessayez.";
   135→   135→   135→  }
   136→   136→   136→  if (/InsecureContext|HTTPS|getUserMedia/i.test(msg)) {
   137→   137→   137→    return "La caméra nécessite une connexion HTTPS sécurisée.";
   138→   138→   138→  }
   139→   139→   139→  return "Impossible d'activer la caméra. Vérifiez votre navigateur et réessayez.";
   140→   140→   140→}

   146→   146→   146→export function ScanTab() {
   147→   147→   147→  // ----- Day selector (persisted) ------------------------------------------
   148→   148→   148→  const [day, setDay] = useState<DayNumber>(readDay);
   149→   149→   149→  useEffect(() => writeLS(LS_DAY, day), [day]);
   150→   150→   150→
   151→   151→   151→  // ----- Effective pause: automatic by clock, or admin-forced (shared) ----
   152→   152→   152→  const { pause: effective } = useEffectivePause();
   153→   153→   153→  const pauseType: PauseType | null = effective?.pauseType ?? null;
   154→   154→   154→
   155→   155→   155→  // Refs mirroring day/pause so the (once-created) scan callback always sees
   156→   156→   156→  // the latest value even if the user switches mid-session.
   157→   157→   157→  const dayRef = useRef(day);
   158→   158→   158→  const pauseTypeRef = useRef<PauseType | null>(pauseType);
   159→   159→   159→  useEffect(() => {
   160→   160→   160→    dayRef.current = day;
   161→   161→   161→  }, [day]);
   162→   162→   162→  useEffect(() => {
   163→   163→   163→    pauseTypeRef.current = pauseType;
   164→   164→   164→  }, [pauseType]);
   165→   165→   165→
   166→   166→   166→  // ----- Camera state ------------------------------------------------------
   167→   167→   167→  const html5Ref = useRef<Html5Qrcode | null>(null);
   168→   168→   168→  const [cameraReady, setCameraReady] = useState(false);
   169→   169→   169→  const [cameraError, setCameraError] = useState<string | null>(null);
   170→   170→   170→  const [cameras, setCameras] = useState<{ id: string; label: string }[]>([]);
   171→   171→   171→  const [cameraIdx, setCameraIdx] = useState(0);
   172→   172→   172→
   173→   173→   173→  // ----- Scan flow state ---------------------------------------------------
   174→   174→   174→  const scanningRef = useRef(true);
   175→   175→   175→  const lastScanRef = useRef<{ id: string; ts: number }>({ id: "", ts: 0 });
   176→   176→   176→  const inFlightRef = useRef(false);
   177→   177→   177→  const resumeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   178→   178→   178→
   179→   179→   179→  const [overlay, setOverlay] = useState<OverlayState | null>(null);
   180→   180→   180→  const [recent, setRecent] = useState<ScanEntry[]>([]);
   181→   181→   181→  const [sessionCount, setSessionCount] = useState(0);
   182→   182→   182→
   183→   183→   183→  // ----- Pause / resume the camera whenever the overlay toggles -----------
   184→   184→   184→  useEffect(() => {
   185→   185→   185→    const inst = html5Ref.current;
   186→   186→   186→    if (!inst) return;
   187→   187→   187→    try {
   188→   188→   188→      if (overlay) inst.pause(true);
   189→   189→   189→      else inst.resume();
   190→   190→   190→    } catch {
   191→   191→   191→      /* ignore */
   192→   192→   192→    }
   193→   193→   193→  }, [overlay]);
   194→   194→   194→
   195→   195→   195→  // ----- recordPause: POST to the API, build overlay + recent entry -------
   196→   196→   196→  const recordPause = useCallback(async (participantId: string) => {
   197→   197→   197→    if (inFlightRef.current) return;
   198→   198→   198→    const d = dayRef.current;
   199→   199→   199→    const p = pauseTypeRef.current;
   200→   200→   200→    if (!p) return; // pause type not known yet — ignore the scan
   201→   201→   201→    inFlightRef.current = true;
   202→   202→   202→    scanningRef.current = false;
   203→   203→   203→
   204→   204→   204→    // Show loading overlay immediately so the user sees feedback.
   205→   205→   205→    setOverlay({
   206→   206→   206→      state: "loading",
   207→   207→   207→      participantId,
   208→   208→   208→      day: d,
   209→   209→   209→      pauseType: p,
   210→   210→   210→    });
   211→   211→   211→
   212→   212→   212→    let result: OverlayState;
   213→   213→   213→    let status: ScanStatus;
   214→   214→   214→
   215→   215→   215→    try {
   216→   216→   216→      const res = await fetch("/api/sonafi/pause/record", {
   217→   217→   217→        method: "POST",
   218→   218→   218→        headers: { "Content-Type": "application/json" },
   219→   219→   219→        body: JSON.stringify({ participantId, day: d, pauseType: p }),
   220→   220→   220→      });
   221→   221→   221→      const data = (await res.json().catch(() => ({}))) as {
   222→   222→   222→        success?: boolean;
   223→   223→   223→        participant?: { id: string; name: string };
   224→   224→   224→        pause?: { timestamp: string };
   225→   225→   225→      };
   226→   226→   226→
   227→   227→   227→      if (res.ok && data?.success) {
   228→   228→   228→        result = {
   229→   229→   229→          state: "success",
   230→   230→   230→          participantId: data.participant?.id ?? participantId,
   231→   231→   231→          participantName: data.participant?.name,
   232→   232→   232→          day: d,
   233→   233→   233→          pauseType: p,
   234→   234→   234→          timestamp: data.pause?.timestamp,
   235→   235→   235→        };
   236→   236→   236→        status = "validated";
   237→   237→   237→        navigator.vibrate?.(120);
   238→   238→   238→      } else if (res.status === 409) {
   239→   239→   239→        result = {
   240→   240→   240→          state: "duplicate",
   241→   241→   241→          participantId: data.participant?.id ?? participantId,
   242→   242→   242→          participantName: data.participant?.name,
   243→   243→   243→          day: d,
   244→   244→   244→          pauseType: p,
   245→   245→   245→        };
   246→   246→   246→        status = "duplicate";
   247→   247→   247→        navigator.vibrate?.([60, 60, 60]);
   248→   248→   248→      } else if (res.status === 404) {
   249→   249→   249→        result = {
   250→   250→   250→          state: "unknown",
   251→   251→   251→          participantId,
   252→   252→   252→          day: d,
   253→   253→   253→          pauseType: p,
   254→   254→   254→        };
   255→   255→   255→        status = "unknown";
   256→   256→   256→        navigator.vibrate?.([60, 60, 60]);
   257→   257→   257→      } else {
   258→   258→   258→        result = { state: "error", participantId, day: d, pauseType: p };
   259→   259→   259→        status = "error";
   260→   260→   260→      }
   261→   261→   261→    } catch {
   262→   262→   262→      result = { state: "error", participantId, day: d, pauseType: p };
   263→   263→   263→      status = "error";
   264→   264→   264→    }
   265→   265→   265→
   266→   266→   266→    // Show the final result.
   267→   267→   267→    setOverlay(result);
   268→   268→   268→
   269→   269→   269→    // Push a recent-scans entry.
   270→   270→   270→    setRecent((prev) =>
   271→   271→   271→      [
   272→   272→   272→        {
   273→   273→   273→          id:
   274→   274→   274→            typeof crypto !== "undefined" && "randomUUID" in crypto
   275→   275→   275→              ? crypto.randomUUID()
   276→   276→   276→              : `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   277→   277→   277→          participantId: result.participantId,
   278→   278→   278→          participantName: result.participantName,
   279→   279→   279→          day: d,
   280→   280→   280→          pauseType: p,
   281→   281→   281→          timestamp: result.timestamp ?? new Date().toISOString(),
   282→   282→   282→          status,
   283→   283→   283→        } as ScanEntry,
   284→   284→   284→        ...prev,
   285→   285→   285→      ].slice(0, MAX_RECENT),
   286→   286→   286→    );
   287→   287→   287→    setSessionCount((c) => c + 1);
   288→   288→   288→
   289→   289→   289→    // Dismiss the overlay and resume scanning after OVERLAY_MS.
   290→   290→   290→    if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   291→   291→   291→    resumeTimerRef.current = setTimeout(() => {
   292→   292→   292→      setOverlay(null);
   293→   293→   293→      scanningRef.current = true;
   294→   294→   294→      inFlightRef.current = false;
   295→   295→   295→    }, OVERLAY_MS);
   296→   296→   296→  }, []);
   297→   297→   297→
   298→   298→   298→  // ----- handleScan: the callback passed to html5-qrcode ------------------
   299→   299→   299→  const handleScan = useCallback(
   300→   300→   300→    (decoded: string) => {
   301→   301→   301→      if (!scanningRef.current) return;
   302→   302→   302→      if (inFlightRef.current) return;
   303→   303→   303→      const now = Date.now();
   304→   304→   304→      const last = lastScanRef.current;
   305→   305→   305→      if (last.id === decoded && now - last.ts < DEBOUNCE_MS) return;
   306→   306→   306→      lastScanRef.current = { id: decoded, ts: now };
   307→   307→   307→      void recordPause(decoded);
   308→   308→   308→    },
   309→   309→   309→    [recordPause],
   310→   310→   310→  );
   311→   311→   311→
   312→   312→   312→  // ----- startCamera: dynamic-import html5-qrcode, prefer back camera -----
   313→   313→   313→  const startCamera = useCallback(
   314→   314→   314→    async (preferredIdx?: number) => {
   315→   315→   315→      setCameraError(null);
   316→   316→   316→      setCameraReady(false);
   317→   317→   317→
   318→   318→   318→      try {
   319→   319→   319→        const { Html5Qrcode } = await import("html5-qrcode");
   320→   320→   320→
   321→   321→   321→        // Tear down any previous instance before starting again.
   322→   322→   322→        if (html5Ref.current) {
   323→   323→   323→          try {
   324→   324→   324→            if (html5Ref.current.isScanning) await html5Ref.current.stop();
   325→   325→   325→          } catch {
   326→   326→   326→            /* ignore */
   327→   327→   327→          }
   328→   328→   328→          try {
   329→   329→   329→            html5Ref.current.clear();
   330→   330→   330→          } catch {
   331→   331→   331→            /* ignore */
   332→   332→   332→          }
   333→   333→   333→          html5Ref.current = null;
   334→   334→   334→        }
   335→   335→   335→
   336→   336→   336→        const instance = new Html5Qrcode(QR_READER_ID, {
   337→   337→   337→          verbose: false,
   338→   338→   338→        });
   339→   339→   339→        html5Ref.current = instance;
   340→   340→   340→
   341→   341→   341→        // Enumerate cameras (best-effort — falls back to facingMode).
   342→   342→   342→        let cams: { id: string; label: string }[] = [];
   343→   343→   343→        try {
   344→   344→   344→          cams = await Html5Qrcode.getCameras();
   345→   345→   345→          if (cams.length > 0) setCameras(cams);
   346→   346→   346→        } catch {
   347→   347→   347→          /* ignore — will use facingMode */
   348→   348→   348→        }
   349→   349→   349→
   350→   350→   350→        const idx =
   351→   351→   351→          preferredIdx != null ? preferredIdx : Math.min(cameraIdx, cams.length);
   352→   352→   352→        const chosen = cams[idx];
   353→   353→   353→
   354→   354→   354→        const config = {
   355→   355→   355→          fps: 10,
   356→   356→   356→          qrbox: (vw: number, vh: number) => {
   357→   357→   357→            const minEdge = Math.min(vw, vh);
   358→   358→   358→            const size = Math.max(160, Math.floor(minEdge * 0.7));
   359→   359→   359→            return { width: size, height: size };
   360→   360→   360→          },
   361→   361→   361→        };
   362→   362→   362→
   363→   363→   363→        const startConfig = chosen
   364→   364→   364→          ? { deviceId: { exact: chosen.id } }
   365→   365→   365→          : { facingMode: { ideal: "environment" as const } };
   366→   366→   366→
   367→   367→   367→        await instance.start(startConfig, config, handleScan, () => {
   368→   368→   368→          /* per-frame decode errors are noisy — ignore */
   369→   369→   369→        });
   370→   370→   370→
   371→   371→   371→        setCameraReady(true);
   372→   372→   372→      } catch (err) {
   373→   373→   373→        setCameraError(friendlyCameraError(err));
   374→   374→   374→        setCameraReady(false);
   375→   375→   375→      }
   376→   376→   376→    },
   377→   377→   377→    [cameraIdx, handleScan],
   378→   378→   378→  );
   379→   379→   379→
   380→   380→   380→  // ----- mount: auto-start camera; unmount: stop + clear -------------------
   381→   381→   381→  useEffect(() => {
   382→   382→   382→    void startCamera();
   383→   383→   383→    return () => {
   384→   384→   384→      if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   385→   385→   385→      const inst = html5Ref.current;
   386→   386→   386→      html5Ref.current = null;
   387→   387→   387→      scanningRef.current = false;
   388→   388→   388→      inFlightRef.current = false;
   389→   389→   389→      void (async () => {
   390→   390→   390→        if (!inst) return;
   391→   391→   391→        try {
   392→   392→   392→          if (inst.isScanning) await inst.stop();
   393→   393→   393→        } catch {
   394→   394→   394→          /* ignore */
   395→   395→   395→        }
   396→   396→   396→        try {
   397→   397→   397→          inst.clear();
   398→   398→   398→        } catch {
   399→   399→   399→          /* ignore */
   400→   400→   400→        }
   401→   401→   401→      })();
   402→   402→   402→    };
   403→   403→   403→    // We deliberately start only once on mount — startCamera is captured at
   404→   404→   404→    // first render, which is exactly what we want (initial camera).
   405→   405→   405→  }, []);
   406→   406→   406→
   407→   407→   407→  // ----- cycleCamera: switch to next available camera ---------------------
   408→   408→   408→  const cycleCamera = useCallback(async () => {
   409→   409→   409→    if (cameras.length < 2) return;
   410→   410→   410→    const next = (cameraIdx + 1) % cameras.length;
   411→   411→   411→    setCameraIdx(next);
   412→   412→   412→    await startCamera(next);
   413→   413→   413→  }, [cameras.length, cameraIdx, startCamera]);
   414→   414→   414→
   415→   415→   415→  // -----------------------------------------------------------------------
   416→   416→   416→  // Render
   417→   417→   417→  // -----------------------------------------------------------------------
   418→   418→   418→  return (
   419→   419→   419→    <div className="flex flex-col gap-5">
   420→   420→   420→      {/* Day selector */}
   421→   421→   421→      <section aria-label="Choix du jour">
   422→   422→   422→        <h2 className="mb-2 text-sm font-semibold text-muted-foreground">Jour</h2>
   423→   423→   423→        <div className="grid grid-cols-3 gap-2">
   424→   424→   424→          {DAYS.map((d) => {
   425→   425→   425→            const active = d === day;
   426→   426→   426→            return (
   427→   427→   427→              <button
   428→   428→   428→                key={d}
