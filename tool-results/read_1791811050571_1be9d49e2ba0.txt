    97→// ---------------------------------------------------------------------------
    98→// localStorage helpers (sanitised — never trust what's in there)
    99→// ---------------------------------------------------------------------------
   100→
   101→function readDay(): DayNumber {
   102→  if (typeof window === "undefined") return 1;
   103→  try {
   104→    const raw = window.localStorage.getItem(LS_DAY);
   105→    const n = raw == null ? NaN : Number(JSON.parse(raw));
   106→    if (n === 1 || n === 2 || n === 3) return n;
   107→  } catch {
   108→    /* ignore */
   109→  }
   110→  return 1;
   111→}
   112→
   113→function writeLS(key: string, value: unknown): void {
   114→  try {
   115→    window.localStorage.setItem(key, JSON.stringify(value));
   116→  } catch {
   117→    /* ignore quota / privacy errors */
   118→  }
   119→}
   120→
   121→// ---------------------------------------------------------------------------
   122→// Friendly camera error translation
   123→// ---------------------------------------------------------------------------
   124→
   125→function friendlyCameraError(err: unknown): string {
   126→  const msg = err instanceof Error ? err.message : String(err);
   127→  if (/NotAllowedError|PermissionDenied/i.test(msg)) {
   128→    return "Accès à la caméra refusé. Autorisez la caméra dans les réglages du navigateur, puis réessayez.";
   129→  }
   130→  if (/NotFoundError|DevicesNotFoundError|CameraNotFound/i.test(msg)) {
   131→    return "Aucune caméra détectée sur cet appareil.";
   132→  }
   133→  if (/NotReadableError|TrackStartError|NotReadable/i.test(msg)) {
   134→    return "La caméra est déjà utilisée par une autre application. Fermez-la, puis réessayez.";
   135→  }
   136→  if (/InsecureContext|HTTPS|getUserMedia/i.test(msg)) {
   137→    return "La caméra nécessite une connexion HTTPS sécurisée.";
   138→  }
   139→  return "Impossible d'activer la caméra. Vérifiez votre navigateur et réessayez.";
   140→}

   142→// ---------------------------------------------------------------------------
   143→// Component
   144→// ---------------------------------------------------------------------------
   145→
   146→export function ScanTab() {
   147→  // ----- Day selector (persisted) ------------------------------------------
   148→  const [day, setDay] = useState<DayNumber>(readDay);
   149→  useEffect(() => writeLS(LS_DAY, day), [day]);
   150→
   151→  // ----- Effective pause: automatic by clock, or admin-forced (shared) ----
   152→  const { pause: effective } = useEffectivePause();
   153→  const pauseType: PauseType | null = effective?.pauseType ?? null;
   154→
   155→  // Refs mirroring day/pause so the (once-created) scan callback always sees
   156→  // the latest value even if the user switches mid-session.
   157→  const dayRef = useRef(day);
   158→  const pauseTypeRef = useRef<PauseType | null>(pauseType);
   159→  useEffect(() => {
   160→    dayRef.current = day;
   161→  }, [day]);
   162→  useEffect(() => {
   163→    pauseTypeRef.current = pauseType;
   164→  }, [pauseType]);
   165→
   166→  // ----- Camera state ------------------------------------------------------
   167→  const html5Ref = useRef<Html5Qrcode | null>(null);
   168→  const [cameraReady, setCameraReady] = useState(false);
   169→  const [cameraError, setCameraError] = useState<string | null>(null);
   170→  const [cameras, setCameras] = useState<{ id: string; label: string }[]>([]);
   171→  const [cameraIdx, setCameraIdx] = useState(0);
   172→
   173→  // ----- Scan flow state ---------------------------------------------------
   174→  const scanningRef = useRef(true);
   175→  const lastScanRef = useRef<{ id: string; ts: number }>({ id: "", ts: 0 });
   176→  const inFlightRef = useRef(false);
   177→  const resumeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   178→
   179→  const [overlay, setOverlay] = useState<OverlayState | null>(null);
   180→  const [recent, setRecent] = useState<ScanEntry[]>([]);
   181→  const [sessionCount, setSessionCount] = useState(0);
   182→
   183→  // ----- Pause / resume the camera whenever the overlay toggles -----------
   184→  useEffect(() => {
   185→    const inst = html5Ref.current;
   186→    if (!inst) return;
   187→    try {
   188→      if (overlay) inst.pause(true);
   189→      else inst.resume();
   190→    } catch {
   191→      /* ignore */
   192→    }
   193→  }, [overlay]);
   194→
   195→  // ----- recordPause: POST to the API, build overlay + recent entry -------
   196→  const recordPause = useCallback(async (participantId: string) => {
   197→    if (inFlightRef.current) return;
   198→    const d = dayRef.current;
   199→    const p = pauseTypeRef.current;
   200→    if (!p) return; // pause type not known yet — ignore the scan
   201→    inFlightRef.current = true;
   202→    scanningRef.current = false;
   203→
   204→    // Show loading overlay immediately so the user sees feedback.
   205→    setOverlay({
   206→      state: "loading",
   207→      participantId,
   208→      day: d,
   209→      pauseType: p,
   210→    });
   211→
   212→    let result: OverlayState;
   213→    let status: ScanStatus;
   214→
   215→    try {
   216→      const res = await fetch("/api/sonafi/pause/record", {
   217→        method: "POST",
   218→        headers: { "Content-Type": "application/json" },
   219→        body: JSON.stringify({ participantId, day: d, pauseType: p }),
   220→      });
   221→      const data = (await res.json().catch(() => ({}))) as {
   222→        success?: boolean;
   223→        participant?: { id: string; name: string };
   224→        pause?: { timestamp: string };
   225→      };
   226→
   227→      if (res.ok && data?.success) {
   228→        result = {
   229→          state: "success",
   230→          participantId: data.participant?.id ?? participantId,
   231→          participantName: data.participant?.name,
   232→          day: d,
   233→          pauseType: p,
   234→          timestamp: data.pause?.timestamp,
   235→        };
   236→        status = "validated";
   237→        navigator.vibrate?.(120);
   238→      } else if (res.status === 409) {
   239→        result = {
   240→          state: "duplicate",
   241→          participantId: data.participant?.id ?? participantId,
   242→          participantName: data.participant?.name,
   243→          day: d,
   244→          pauseType: p,
   245→        };
   246→        status = "duplicate";
   247→        navigator.vibrate?.([60, 60, 60]);
   248→      } else if (res.status === 404) {
   249→        result = {
   250→          state: "unknown",
   251→          participantId,
   252→          day: d,
   253→          pauseType: p,
   254→        };
   255→        status = "unknown";
   256→        navigator.vibrate?.([60, 60, 60]);
   257→      } else {
   258→        result = { state: "error", participantId, day: d, pauseType: p };
   259→        status = "error";
   260→      }
   261→    } catch {
   262→      result = { state: "error", participantId, day: d, pauseType: p };
   263→      status = "error";
   264→    }
   265→
   266→    // Show the final result.
   267→    setOverlay(result);
   268→
   269→    // Push a recent-scans entry.
   270→    setRecent((prev) =>
   271→      [
   272→        {
   273→          id:
   274→            typeof crypto !== "undefined" && "randomUUID" in crypto
   275→              ? crypto.randomUUID()
   276→              : `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   277→          participantId: result.participantId,
   278→          participantName: result.participantName,
   279→          day: d,
   280→          pauseType: p,
   281→          timestamp: result.timestamp ?? new Date().toISOString(),
   282→          status,
   283→        } as ScanEntry,
   284→        ...prev,
   285→      ].slice(0, MAX_RECENT),
   286→    );
   287→    setSessionCount((c) => c + 1);
   288→
   289→    // Dismiss the overlay and resume scanning after OVERLAY_MS.
   290→    if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   291→    resumeTimerRef.current = setTimeout(() => {
   292→      setOverlay(null);
   293→      scanningRef.current = true;
   294→      inFlightRef.current = false;
   295→    }, OVERLAY_MS);
   296→  }, []);
   297→
   298→  // ----- handleScan: the callback passed to html5-qrcode ------------------
   299→  const handleScan = useCallback(
   300→    (decoded: string) => {
   301→      if (!scanningRef.current) return;
   302→      if (inFlightRef.current) return;
   303→      const now = Date.now();
   304→      const last = lastScanRef.current;
   305→      if (last.id === decoded && now - last.ts < DEBOUNCE_MS) return;
   306→      lastScanRef.current = { id: decoded, ts: now };
   307→      void recordPause(decoded);
   308→    },
   309→    [recordPause],
   310→  );
   311→
   312→  // ----- startCamera: dynamic-import html5-qrcode, prefer back camera -----
   313→  const startCamera = useCallback(
   314→    async (preferredIdx?: number) => {
   315→      setCameraError(null);
   316→      setCameraReady(false);
   317→
   318→      try {
   319→        const { Html5Qrcode } = await import("html5-qrcode");
   320→
   321→        // Tear down any previous instance before starting again.
   322→        if (html5Ref.current) {
   323→          try {
   324→            if (html5Ref.current.isScanning) await html5Ref.current.stop();
   325→          } catch {
   326→            /* ignore */
   327→          }
   328→          try {
   329→            html5Ref.current.clear();
   330→          } catch {
   331→            /* ignore */
   332→          }
   333→          html5Ref.current = null;
   334→        }
   335→
   336→        const instance = new Html5Qrcode(QR_READER_ID, {
   337→          verbose: false,
   338→        });
   339→        html5Ref.current = instance;
   340→
   341→        // Enumerate cameras (best-effort — falls back to facingMode).
   342→        let cams: { id: string; label: string }[] = [];
   343→        try {
   344→          cams = await Html5Qrcode.getCameras();
   345→          if (cams.length > 0) setCameras(cams);
   346→        } catch {
   347→          /* ignore — will use facingMode */
   348→        }
   349→
   350→        const idx =
   351→          preferredIdx != null ? preferredIdx : Math.min(cameraIdx, cams.length);
   352→        const chosen = cams[idx];
   353→
   354→        const config = {
   355→          fps: 10,
   356→          qrbox: (vw: number, vh: number) => {
   357→            const minEdge = Math.min(vw, vh);
   358→            const size = Math.max(160, Math.floor(minEdge * 0.7));
   359→            return { width: size, height: size };
   360→          },
   361→        };
   362→
   363→        const startConfig = chosen
   364→          ? { deviceId: { exact: chosen.id } }
   365→          : { facingMode: { ideal: "environment" as const } };
   366→
   367→        await instance.start(startConfig, config, handleScan, () => {
   368→          /* per-frame decode errors are noisy — ignore */
   369→        });
   370→
   371→        setCameraReady(true);
   372→      } catch (err) {
   373→        setCameraError(friendlyCameraError(err));
   374→        setCameraReady(false);
   375→      }
   376→    },
   377→    [cameraIdx, handleScan],
   378→  );
   379→
   380→  // ----- mount: auto-start camera; unmount: stop + clear -------------------
   381→  useEffect(() => {
   382→    void startCamera();
   383→    return () => {
   384→      if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current);
   385→      const inst = html5Ref.current;
   386→      html5Ref.current = null;
   387→      scanningRef.current = false;
   388→      inFlightRef.current = false;
   389→      void (async () => {
   390→        if (!inst) return;
   391→        try {
   392→          if (inst.isScanning) await inst.stop();
   393→        } catch {
   394→          /* ignore */
   395→        }
   396→        try {
   397→          inst.clear();
   398→        } catch {
   399→          /* ignore */
   400→        }
   401→      })();
   402→    };
   403→    // We deliberately start only once on mount — startCamera is captured at
   404→    // first render, which is exactly what we want (initial camera).
   405→  }, []);
   406→
   407→  // ----- cycleCamera: switch to next available camera ---------------------
   408→  const cycleCamera = useCallback(async () => {
   409→    if (cameras.length < 2) return;
   410→    const next = (cameraIdx + 1) % cameras.length;
   411→    setCameraIdx(next);
   412→    await startCamera(next);
   413→  }, [cameras.length, cameraIdx, startCamera]);
   414→
   415→  // -----------------------------------------------------------------------
   416→  // Render
   417→  // -----------------------------------------------------------------------
   418→  return (
   419→    <div className="flex flex-col gap-5">
   420→      {/* Day selector */}
   421→      <section aria-label="Choix du jour">
   422→        <h2 className="mb-2 text-sm font-semibold text-muted-foreground">Jour</h2>
   423→        <div className="grid grid-cols-3 gap-2">
   424→          {DAYS.map((d) => {
   425→            const active = d === day;
   426→            return (
   427→              <button
   428→                key={d}
   429→                type="button"
   430→                onClick={() => setDay(d)}
   431→                aria-pressed={active}
   432→                className={cn(
   433→                  "flex h-12 items-center justify-center rounded-xl border text-sm font-semibold transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400 focus-visible:ring-offset-2",
   434→                  active
   435→                    ? "border-emerald-500 bg-emerald-500 text-white shadow-sm"
   436→                    : "border-border bg-background text-foreground hover:bg-emerald-50",
   437→                )}
   438→              >
   439→                {DAY_LABELS[d]}
   440→              </button>
   441→            );
   442→          })}
   443→        </div>
   444→      </section>
   445→
   446→      {/* Active pause — automatic by clock, or forced by admin */}
   447→      <section aria-label="Pause en cours">
   448→        <PauseBadge pause={effective} />
   449→      </section>
   450→
   451→      {/* Camera */}
   452→      <section aria-label="Caméra de scan">
   453→        <div className="mb-2 flex items-center justify-between gap-3">
   454→          <h2 className="text-sm font-semibold text-muted-foreground">Caméra</h2>
   455→          <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground">
   456→            <ScanLine className="h-3.5 w-3.5" />
   457→            {sessionCount} scan{sessionCount === 1 ? "" : "s"} cette session
   458→          </span>
   459→        </div>
   460→
   461→        <Card className="overflow-hidden border-emerald-100 py-0">
   462→          <CardContent className="p-3 sm:p-4">
   463→            <div className="relative mx-auto aspect-square w-full max-w-md overflow-hidden rounded-2xl border border-border bg-slate-900">
   464→              {/* html5-qrcode injects the <video> here */}
   465→              <div
   466→                id={QR_READER_ID}
   467→                className="absolute inset-0 [&>video]:h-full [&>video]:w-full [&>video]:object-cover [&>video]:bg-slate-900"
   468→              />
   469→
   470→              {/* Scan reticle overlay */}
   471→              {cameraReady && !cameraError && (
   472→                <div className="pointer-events-none absolute inset-0 flex items-center justify-center">
   473→                  <div className="relative h-[62%] w-[62%]">
   474→                    <span className="absolute left-0 top-0 h-8 w-8 rounded-tl-lg border-l-4 border-t-4 border-white/90" />
   475→                    <span className="absolute right-0 top-0 h-8 w-8 rounded-tr-lg border-r-4 border-t-4 border-white/90" />
   476→                    <span className="absolute bottom-0 left-0 h-8 w-8 rounded-bl-lg border-b-4 border-l-4 border-white/90" />
   477→                    <span className="absolute bottom-0 right-0 h-8 w-8 rounded-br-lg border-b-4 border-r-4 border-white/90" />
   478→                    <motion.span
   479→                      className="absolute inset-x-4 top-1/2 h-0.5 rounded-full bg-emerald-400 shadow-[0_0_8px_2px_rgba(16,185,129,0.65)]"
   480→                      animate={{ y: [-90, 90, -90] }}
   481→                      transition={{
   482→                        duration: 2.4,
   483→                        repeat: Infinity,
   484→                        ease: "easeInOut",
   485→                      }}
   486→                    />
   487→                  </div>
   488→                </div>
   489→              )}
   490→
   491→              {/* Initialising spinner */}
   492→              {!cameraReady && !cameraError && (
   493→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 text-white/90">
   494→                  <Loader2 className="h-8 w-8 animate-spin" />
   495→                  <p className="text-sm">Activation de la caméra…</p>
   496→                </div>
   497→              )}
   498→
   499→              {/* Camera error card */}
   500→              {cameraError && (
   501→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-3 bg-slate-900/95 p-6 text-center text-white">
   502→                  <AlertTriangle className="h-10 w-10 text-rose-300" />
   503→                  <p className="max-w-xs text-sm leading-relaxed text-white/90">
   504→                    {cameraError}
   505→                  </p>
   506→                  <Button
   507→                    type="button"
   508→                    size="sm"
   509→                    onClick={() => void startCamera()}
   510→                    className="bg-emerald-500 text-white hover:bg-emerald-600"
   511→                  >
   512→                    <RefreshCw className="h-4 w-4" />
   513→                    Réessayer
   514→                  </Button>
   515→                </div>
   516→              )}
   517→
   518→              {/* Camera switch */}
   519→              {cameraReady && cameras.length > 1 && (
   520→                <Button
   521→                  type="button"
   522→                  size="sm"
   523→                  variant="secondary"
   524→                  onClick={() => void cycleCamera()}
   525→                  className="absolute right-2 top-2 z-10 h-8 gap-1 bg-black/55 text-white backdrop-blur hover:bg-black/70"
   526→                >
   527→                  <RotateCcw className="h-3.5 w-3.5" />
   528→                  Caméra
   529→                </Button>
   530→              )}
   531→            </div>
   532→
   533→            {/* Status row */}
   534→            <div className="mt-3 flex items-center justify-between text-xs text-muted-foreground">
   535→              <span className="inline-flex items-center gap-1.5">
   536→                <span
   537→                  className={cn(
   538→                    "h-2 w-2 rounded-full",
   539→                    cameraReady
   540→                      ? "animate-pulse bg-emerald-500"
   541→                      : "bg-muted-foreground/40",
   542→                  )}
   543→                />
   544→                {cameraReady
   545→                  ? "Caméra active"
   546→                  : cameraError
   547→                    ? "Caméra indisponible"
   548→                    : "Démarrage…"}
   549→              </span>
   550→              <span className="inline-flex items-center gap-1.5">
   551→                {pauseType
   552→                  ? `${PAUSE_META[pauseType].emoji} ${PAUSE_META[pauseType].label}`
   553→                  : "Pause…"}{" "}·{" "}
   554→                {DAY_LABELS[day]}
   555→              </span>
   556→            </div>
   557→          </CardContent>
   558→        </Card>
   559→      </section>
   560→
   561→      {/* Recent scans */}
   562→      <section aria-label="Scans récents">
   563→        <div className="mb-2 flex items-center justify-between">
   564→          <h2 className="text-sm font-semibold text-muted-foreground">
   565→            Scans récents
   566→          </h2>
   567→          <span className="text-xs text-muted-foreground">
   568→            {recent.length}/{MAX_RECENT}
   569→          </span>
   570→        </div>
   571→
   572→        {recent.length === 0 ? (
   573→          <Card className="border-dashed bg-muted/30 py-0">
   574→            <CardContent className="flex items-center justify-center py-8 text-sm text-muted-foreground">
   575→              Aucun scan pour le moment
   576→            </CardContent>
   577→          </Card>
   578→        ) : (
   579→          <div
   580→            className="max-h-72 overflow-y-auto pr-1
   581→              [&::-webkit-scrollbar]:w-2
   582→              [&::-webkit-scrollbar-thumb]:rounded-full
   583→              [&::-webkit-scrollbar-thumb]:bg-muted-foreground/30
   584→              [&::-webkit-scrollbar-track]:bg-transparent"
   585→          >
   586→            <ul className="flex flex-col gap-2">
   587→              {recent.map((e) => (
   588→                <li key={e.id}>
   589→                  <Card className="border-border/70 py-0">
   590→                    <CardContent className="flex items-center gap-3 px-3 py-2.5">

   727→// ---------------------------------------------------------------------------
   728→// ResultOverlay — full-screen verdict shown after each scan
   729→// ---------------------------------------------------------------------------
   730→
   731→function ResultOverlay({ overlay }: { overlay: OverlayState | null }) {
   732→  const theme: OverlayTheme | null = (() => {
   733→    if (!overlay) return null;
   734→    const meta = `${PAUSE_META[overlay.pauseType].emoji} ${PAUSE_META[overlay.pauseType].label} · ${DAY_LABELS[overlay.day]}`;
   735→    switch (overlay.state) {
   736→      case "loading":
   737→        return {
   738→          bg: "bg-slate-900/95",
   739→          icon: <Loader2 className="h-16 w-16 animate-spin text-white" />,
   740→          headline: "Vérification…",
   741→          sub: overlay.participantId,
   742→        };
   743→      case "success":
   744→        return {
   745→          bg: "bg-emerald-600",
   746→          icon: <Check className="h-20 w-20 text-white" strokeWidth={3} />,
   747→          headline: "Pause validée",
   748→          sub: overlay.participantName ?? "Participant",
   749→          meta,
   750→        };
   751→      case "duplicate":
   752→        return {
   753→          bg: "bg-rose-600",
   754→          icon: <X className="h-20 w-20 text-white" strokeWidth={3} />,
   755→          headline: "Déjà servi(e)",
   756→          sub: overlay.participantName ?? overlay.participantId,
   757→          meta,
   758→        };
   759→      case "unknown":
   760→        return {
   761→          bg: "bg-amber-500",
   762→          icon: <AlertTriangle className="h-20 w-20 text-white" strokeWidth={2.5} />,
   763→          headline: "Badge non reconnu",
   764→          sub: overlay.participantId,
   765→        };
   766→      case "error":
   767→        return {
   768→          bg: "bg-slate-700",
   769→          icon: <WifiOff className="h-20 w-20 text-white" strokeWidth={2.5} />,
   770→          headline: "Erreur, réessayez",
   771→          sub: "Vérifiez votre connexion",
   772→        };
   773→    }
   774→  })();
   775→
   776→  return (
   777→    <AnimatePresence>
   778→      {overlay && theme && (
   779→        <motion.div
   780→          initial={{ opacity: 0 }}
   781→          animate={{ opacity: 1 }}
   782→          exit={{ opacity: 0 }}
   783→          transition={{ duration: 0.18 }}
   784→          className={cn(
   785→            "fixed inset-0 z-50 flex flex-col items-center justify-center p-8 text-center text-white",
   786→            theme.bg,
   787→          )}
   788→          aria-live="assertive"
   789→          role="status"
   790→        >
   791→          <motion.div
   792→            key={overlay.state}
   793→            initial={{ scale: 0.5, opacity: 0 }}
   794→            animate={{ scale: 1, opacity: 1 }}
   795→            transition={{ type: "spring", stiffness: 320, damping: 22 }}
   796→            className="flex flex-col items-center gap-5"
   797→          >
   798→            {theme.icon}
   799→            <h2 className="text-3xl font-bold tracking-tight sm:text-4xl">
   800→              {theme.headline}
   801→            </h2>
   802→            {theme.sub && (
   803→              <p className="max-w-md break-words text-xl font-medium opacity-95">
   804→                {theme.sub}
   805→              </p>
   806→            )}
   807→            {theme.meta && <p className="text-lg opacity-90">{theme.meta}</p>}
   808→          </motion.div>
   809→        </motion.div>
   810→      )}
   811→    </AnimatePresence>
   812→  );
   813→}
